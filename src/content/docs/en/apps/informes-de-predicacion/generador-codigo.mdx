---
title: Configuration code generator
description: Tool to register a new congregation and generate the code that users enter in the app.
---

import { Aside } from '@astrojs/starlight/components';

Creates the project code that the user enters on the first screen of the app.
When saving, the data is registered in the database of the app's base project.
The user only needs the code: they do not receive or handle Firebase credentials.

<Aside type="caution">
This tool writes directly to the database. Verify the data before saving.
A code that is already published can be used by anyone who knows it.
</Aside>

<script type="module">{`
  // ─── Base app project Firebase configuration ───────────────────────────────
  // Replace these values with those from the Field Service Reports Firebase project
  const FIREBASE_CONFIG = {
    apiKey:            "AIzaSyBGtfasTpBWfO6fckcuUkOb_3wpl0EiJeY",
    authDomain:        "informes-predicacion.firebaseapp.com",
    projectId:         "informes-predicacion",
    storageBucket:     "informes-predicacion.firebasestorage.app",
    messagingSenderId: "631448576676",
    appId:             "1:631448576676:web:4663c2cdde156363e97c31"
  };
  const COLLECTION = "proyectos";
  // ─────────────────────────────────────────────────────────────────────────

  import { initializeApp, getApps } from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-app.js';
  import { getFirestore, doc, setDoc, query, collection, where, getDocs }
    from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-firestore.js';
  import { getAuth, signInAnonymously }
    from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-auth.js';

  const app = getApps().length ? getApps()[0] : initializeApp(FIREBASE_CONFIG);
  const db   = getFirestore(app);
  const auth = getAuth(app);

  // ── Utilities ─────────────────────────────────────────────────────────────

  function generarCodigo() {
    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
    return Array.from({ length: 6 }, () => chars[Math.floor(Math.random() * chars.length)]).join('');
  }

  function leerFormulario() {
    return {
      codigo:        document.getElementById('g-codigo').value.trim().toUpperCase(),
      projectName:   document.getElementById('g-nombre').value.trim(),
      collectionName: document.getElementById('g-coleccion').value.trim() || 'informes',
      editBaseUrl:   document.getElementById('g-editurl').value.trim(),
      groups:        document.getElementById('g-grupos').value
                       .split(',').map(g => g.trim()).filter(Boolean),
      firebaseConfig: {
        apiKey:        document.getElementById('g-apikey').value.trim(),
        applicationId: document.getElementById('g-appid').value.trim(),
        projectId:     document.getElementById('g-projectid').value.trim()
      }
    };
  }

  function validar(datos) {
    const errs = [];
    if (!datos.codigo)                        errs.push('Code is required');
    if (!datos.projectName)                   errs.push('Project name is required');
    if (!datos.firebaseConfig.apiKey)         errs.push('API Key is required');
    if (!datos.firebaseConfig.applicationId)  errs.push('Application ID is required');
    if (!datos.firebaseConfig.projectId)      errs.push('Project ID is required');
    if (datos.groups.length === 0)            errs.push('Add at least one group');
    return errs;
  }

  function setStatus(msg, tipo) {
    const el = document.getElementById('gen-status');
    el.textContent = msg;
    el.className = tipo;
  }

  // ── Events ────────────────────────────────────────────────────────────────

  document.getElementById('btn-generar').addEventListener('click', () => {
    document.getElementById('g-codigo').value = generarCodigo();
  });

  document.getElementById('btn-preview').addEventListener('click', () => {
    const datos = leerFormulario();
    const pre = document.getElementById('gen-preview');
    pre.style.display = 'block';
    pre.textContent = JSON.stringify(datos, null, 2);
  });

  document.getElementById('btn-guardar').addEventListener('click', async () => {
    const btn = document.getElementById('btn-guardar');
    const datos = leerFormulario();

    const errores = validar(datos);
    if (errores.length > 0) {
      setStatus(errores.join(' · '), 'error');
      return;
    }

    btn.disabled = true;
    btn.textContent = 'Saving…';
    setStatus('', '');

    try {
      await signInAnonymously(auth);

      // Check if code already exists
      const q = query(collection(db, COLLECTION), where('codigo', '==', datos.codigo));
      const existe = await getDocs(q);
      if (!existe.empty) {
        setStatus(\`The code "\${datos.codigo}" already exists. Generate a different one or edit the existing one.\`, 'error');
        return;
      }

      // Save document using the code as ID for easy reference
      await setDoc(doc(db, COLLECTION, datos.codigo), datos);

      setStatus(\`Code "\${datos.codigo}" saved successfully for "\${datos.projectName}".\`, 'success');
      document.getElementById('gen-preview').style.display = 'none';
    } catch (err) {
      setStatus(\`Error: \${err.message}\`, 'error');
    } finally {
      btn.disabled = false;
      btn.textContent = 'Save to Firebase';
    }
  });

  // Generate code on page load
  document.getElementById('g-codigo').value = generarCodigo();
  document.getElementById('g-coleccion').value = 'informes';
`}</script>
